<objective>
Phase 1 gap-closure plan 16 — close the structural gap identified by BLOCKER 1 of the plan-checker review on plans 10..15: cross-invocation cursor flow.

**The bug.** Today the orchestrator does:
```ts
const runId = makeRunId();                       // fresh per invocation
const runRoot = resolve(RUN_ROOT_REL);
const runDir = resolve(runRoot, runId);          // fresh per invocation
await mkdir(runDir, { recursive: true });
cursor = await readCursor(runDir);               // reads runDir/.cursor.json — EMPTY on every invocation
// ...
await writeCursor(runDir, { lastBrandSlug, lastModelSlug, completedAt }); // writes runDir/.cursor.json
```

Because `runId` is generated from the current clock at the start of every `run()`, the cursor written into `runId-N`'s directory is NEVER read by `runId-N+1`. The cursor's resume contract is structurally broken in production: `cli.ts:27` always passes `resume: true`, but every invocation reads from a fresh empty directory and gets `null`. The plan-13 integration tests pass because they `vi.doMock` `readCursor` to inject a synthetic cursor — they validate the resume LOGIC under a cursor-present assumption, but they do NOT validate that the file written by run N reaches run N+1.

**The fix.** Move the cursor file to a stable, runId-agnostic location: `data/scraped/drom/.cursor.json` at the brand root. The orchestrator computes the path ONCE at the top of `run()`, passes it to every cursor call, and the file persists across `runId` boundaries — so run N+1 reads what run N wrote.

This plan delivers:
1. `server/scrapers/shared/cursor.ts` — change `readCursor(runDir)` → `readCursor(cursorPath)`, same for `writeCursor` and `deleteCursor`. The functions stop deriving the path from `runDir`; the caller owns the path.
2. `server/scrapers/drom/index.ts` — compute `cursorPath = resolve(runRoot, '.cursor.json')` once at the top of `run()`, replace every `readCursor(runDir)` / `writeCursor(runDir, ...)` / `deleteCursor(runDir)` call site with the new path argument.
3. `server/tests/cursor.test.ts` — update existing tests to pass `cursorPath` (e.g. `resolve(runDir, '.cursor.json')`) instead of `runDir`. Tests must continue to pass.
4. `server/tests/drom-integration.test.ts` — add a NEW integration test that:
   - Invokes `drom.run({ resume: false })` with a stub that throws after N models, leaving a real `.cursor.json` on disk at the brand-root path.
   - Asserts the cursor file exists at `data/scraped/drom/.cursor.json` (NOT inside any per-run directory).
   - Invokes `drom.run({ resume: true })` and asserts the second invocation reads the cursor (`report.cursor_resumed === true`).
5. `data/scraped/README.md` — update §"Crash recovery" to reference the new stable cursor path and clarify that the cursor file is shared across run invocations.

```typescript
// server/scrapers/shared/cursor.ts
import { readFile, unlink } from 'node:fs/promises';
import { z } from 'zod';
import { atomicWriteFile } from './atomic-write.js';

/**
 * D-15 brand-boundary cursor.
 *
 * Pitfall 3 trade-off (RESEARCH.md lines 818-826): brand-boundary checkpointing
 * is coarser than ideal — worst-case ~7 hours wasted on a mid-brand crash.
 * Phase 1 keeps this simple; finer-grained cursor is a Phase 1.x candidate.
 *
 * Plan 01-11 (gap-closure): readCursor distinguishes ENOENT from corrupt JSON
 * and validates shape via zod (WR-04 fix).
 *
 * Plan 01-16 (gap-closure): the cursor file path is now an EXTERNAL concern.
 * Callers (the drom orchestrator) own the path and pass it explicitly. The
 * canonical location is `data/scraped/drom/.cursor.json` at the brand root —
 * NOT inside per-run runDirs (which are fresh per invocation and would defeat
 * cross-invocation resume). See data/scraped/README.md §"Crash recovery".
 */

export const CursorSchema = z.object({
  lastBrandSlug: z.string().min(1),
  lastModelSlug: z.string().min(1),
  completedAt: z.string().datetime(),
});
export type Cursor = z.infer<typeof CursorSchema>;

/**
 * Thrown when the cursor file exists but cannot be parsed as a valid Cursor.
 * Distinct from "file absent" (returns null) and "file unreadable for OS
 * reasons" (propagates the underlying NodeJS error). Caller should surface
 * this to the operator and refuse to start a fresh run silently.
 */
export class CorruptCursorError extends Error {
  constructor(message: string, public readonly cause?: unknown) {
    super(message);
    this.name = 'CorruptCursorError';
  }
}

/**
 * Read the cursor file at `cursorPath`. Returns null on ENOENT (fresh start).
 * Throws CorruptCursorError on corrupt JSON or shape mismatch. Propagates
 * other read errors (EACCES, etc.) unchanged.
 */
export async function readCursor(cursorPath: string): Promise<Cursor | null> {
  let raw: string;
  try {
    raw = await readFile(cursorPath, 'utf-8');
  } catch (err) {
    if ((err as NodeJS.ErrnoException).code === 'ENOENT') return null;
    throw err;
  }
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch (err) {
    throw new CorruptCursorError(
      `Cursor at ${cursorPath} is corrupt JSON; refusing to silently restart. ` +
        `Delete the file explicitly to begin a fresh run.`,
      err,
    );
  }
  const result = CursorSchema.safeParse(parsed);
  if (!result.success) {
    throw new CorruptCursorError(
      `Cursor at ${cursorPath} has shape mismatch (expected {lastBrandSlug, lastModelSlug, completedAt}); ` +
        `refusing to silently restart. zod issues: ${JSON.stringify(result.error.issues)}`,
      result.error,
    );
  }
  return result.data;
}

/**
 * Atomically write the cursor to `cursorPath`. The caller is responsible for
 * ensuring the parent directory exists.
 */
export async function writeCursor(cursorPath: string, cursor: Cursor): Promise<void> {
  await atomicWriteFile(cursorPath, JSON.stringify(cursor, null, 2));
}

/**
 * Idempotently delete the cursor file. No-op on ENOENT.
 */
export async function deleteCursor(cursorPath: string): Promise<void> {
  await unlink(cursorPath).catch(() => {
    /* idempotent — no throw if absent */
  });
}
```
