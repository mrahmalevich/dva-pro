    ```js
    // tmp/measure-post-wave1.mjs — Phase 02.1 Wave 2 measurement (ephemeral; delete after run)
    import puppeteer from 'puppeteer';
    import sharp from 'sharp';
    import { readFile, writeFile } from 'node:fs/promises';
    import { resolve } from 'node:path';
    import { PNG } from 'pngjs';
    import pixelmatch from 'pixelmatch';

    const REFERENCE = resolve('.planning/phases/02-redesign-from-screenshot/design-reference.png');
    const VIEWPORT = { width: 1280, height: 4000 };
    const REF_W = 605;
    const REF_H = 1280;
    const DEV_URL = 'http://127.0.0.1:5174/';

    const browser = await puppeteer.launch({
      headless: true,
      args: ['--font-render-hinting=none', '--disable-font-subpixel-positioning'],
    });
    try {
      const page = await browser.newPage();
      await page.setViewport(VIEWPORT);
      await page.goto(DEV_URL, { waitUntil: 'networkidle0', timeout: 30_000 });
      await new Promise(r => setTimeout(r, 1500));

      const screenshotData = await page.screenshot({ type: 'png', fullPage: true });
      const captureBuf = Buffer.isBuffer(screenshotData) ? screenshotData : Buffer.from(screenshotData);

      // Capture height of the full SPA (D-01a band check)
      const meta = await sharp(captureBuf).metadata();
      const fullPageHeightPx = meta.height;
      console.log(JSON.stringify({ fullPageHeightPx }));

      // Save the captured fullPage image for the SUMMARY (gitignored tmp/)
      await writeFile('tmp/landing-post-wave1-fullpage.png', captureBuf);

      // Pixelmatch floor against the reference (D-03b safety check + D-03 threshold tuning)
      const resizedBuf = await sharp(captureBuf)
        .resize(REF_W, REF_H, { kernel: 'cubic', fit: 'fill' })
        .png()
        .toBuffer();
      await writeFile('tmp/landing-post-wave1-resized.png', resizedBuf);

      const actualPng = PNG.sync.read(resizedBuf);
      const expectedPng = PNG.sync.read(await readFile(REFERENCE));
      const diff = new PNG({ width: REF_W, height: REF_H });
      const mismatched = pixelmatch(
        actualPng.data,
        expectedPng.data,
        diff.data,
        REF_W,
        REF_H,
        { threshold: 0.1 },
      );
      const ratio = mismatched / (REF_W * REF_H);
      const floorPct = (ratio * 100).toFixed(2);
      console.log(JSON.stringify({ floor: ratio, floorPct, mismatched, totalPx: REF_W * REF_H }));
    } finally {
      await browser.close();
    }
    ```
