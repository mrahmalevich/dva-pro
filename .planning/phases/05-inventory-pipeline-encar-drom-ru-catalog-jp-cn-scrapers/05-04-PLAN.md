    ```typescript
    import { eq, sql } from 'drizzle-orm';
    import type { Database } from '../../../infra/db.js';
    import type { Queue } from '../../../infra/queue.js';
    import { scraperConfig, scraperRuns } from '../../../db/schema/index.js';
    import type { ScraperSource } from './types.js';

    export const BLOCK_THRESHOLD = 5;
    export const COOLDOWN_HOURS = 24;

    // Captcha keyword regex — covers CN, KR, EN, RU defensive
    const CAPTCHA_PATTERNS = /验证|请稍候|보안 인증|잠시만 기다려 주세요|recaptcha|cf-challenge|prove you('|')re human|robot.{0,10}check|please verify|капча|проверка/i;

    export type BlockSignal = 'thin_body' | 'captcha_keyword' | 'http_403' | 'http_429';

    export function detectBlock(r: { html: string; status: number }): BlockSignal | null {
      if (r.status === 403) return 'http_403';
      if (r.status === 429) return 'http_429';
      if (r.html.length < 1024) return 'thin_body';
      if (CAPTCHA_PATTERNS.test(r.html)) return 'captcha_keyword';
      return null;
    }

    /** Per-run counter. consecutiveBlocks resets on any non-block response. */
    export class BlockCounter {
      private count = 0;
      constructor(private readonly source: ScraperSource, private readonly db: Database, private readonly queue: Queue) {}
      async record(signal: BlockSignal | null): Promise<{ halted: boolean }> {
        if (signal === null) { this.count = 0; return { halted: false }; }
        this.count += 1;
        if (this.count >= BLOCK_THRESHOLD) { await haltSource(this.source, signal, this.db, this.queue); return { halted: true }; }
        return { halted: false };
      }
    }

    export async function haltSource(source: ScraperSource, reason: BlockSignal, db: Database, queue: Queue): Promise<void> {
      await db.update(scraperConfig).set({
        pausedUntil: sql`NOW() + INTERVAL '${sql.raw(String(COOLDOWN_HOURS))} hours'`,
        lastRunStatus: 'blocked',
        lastBlockReason: reason,
      }).where(eq(scraperConfig.source, source));
      await queue.send('email.founder-alert', {
        template: 'scraper-blocked',
        source, reason,
        paused_until: `+${COOLDOWN_HOURS}h`,
      });
    }

    /** Pre-flight check at scrape-run start. Returns true if pausedUntil > NOW(). */
    export async function isPaused(source: ScraperSource, db: Database): Promise<boolean> { /* SELECT paused_until ... */ }
    ```
