    ```typescript
    import * as cheerio from 'cheerio';
    import type { Database } from '../../../infra/db.js';
    import type { NormalizedCar } from '../shared/types.js';
    import { parsePriceCn, parseMileage, mapBody, mapDrive, mapFuel, mapTransmission, buildSpecSummary } from '../shared/normalize.js';
    import { lookupBrandAlias, upsertBrandAlias } from '../shared/aliases.js';

    /** Listing page → ref ids (extracted from /dealer/{shop_id}/{car_id}.html or /owner/{car_id}.html) */
    export function parseListingPage(html: string): Array<{ sourceId: string; sourceUrl: string }> { /* ... */ }

    export function parseDetailPage(html: string): {
      brandCn: string; modelCn: string; year: number; bodyCn: string;
      driveCn: string | null; fuelCn: string | null; transmissionCn: string | null;
      mileageRaw: string; priceRaw: string; imageUrls: string[];
    } { /* ... */ }

    export async function normalizeChe168(
      sourceId: string, sourceUrl: string, detailHtml: string, db: Database
    ): Promise<NormalizedCar> {
      const raw = parseDetailPage(detailHtml);
      // Mandarin → canonical via brand_aliases (CN seed from plan 02 covers ~50 brands)
      let aliasBrand = await lookupBrandAlias(db, { kind: 'brand', altForm: raw.brandCn });
      if (!aliasBrand) {
        // Unmatched on first sight — write a canonical-only row for admin to reconcile
        const canonicalSlug = raw.brandCn.toLowerCase().replace(/[^a-z0-9]/g, '');
        await upsertBrandAlias(db, {
          kind: 'brand',
          canonicalSlug: canonicalSlug || raw.brandCn,
          latinForm: raw.brandCn,
          altForm: raw.brandCn,
          source: 'cn-seed',
        });
        aliasBrand = await lookupBrandAlias(db, { kind: 'brand', altForm: raw.brandCn });
      }
      const brand = aliasBrand?.cyrillicForm ?? aliasBrand?.latinForm ?? raw.brandCn;
      const model = raw.modelCn;

      return {
        source: 'che168', sourceId, sourceUrl,
        brand, model, year: raw.year,
        country: 'cn',
        body: mapBody(raw.bodyCn),
        drive: mapDrive(raw.driveCn ?? ''),
        fuel: mapFuel(raw.fuelCn ?? ''),
        transmission: mapTransmission(raw.transmissionCn ?? ''),
        mileageKm: parseMileage(raw.mileageRaw),
        priceLocal: parsePriceCn(raw.priceRaw) ?? 0n,
        priceLocalCcy: 'CNY',
        specSummary: buildSpecSummary({}),
        imageUrls: raw.imageUrls,
        needsReview: true,
      };
    }
    ```
