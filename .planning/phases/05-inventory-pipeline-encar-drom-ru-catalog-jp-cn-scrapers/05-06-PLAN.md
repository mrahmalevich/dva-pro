    ```typescript
    export type ProxyOpts = { geo: 'kr' | 'cn'; sessionTtlSec?: number; sessionId?: string };

    /** Returns IPRoyal-formatted proxy URL or null if creds unavailable. v1: hardcoded geo.iproyal.com:12321. */
    export function buildProxyUrl(opts: ProxyOpts): string | null {
      const user = process.env.IPROYAL_PROXY_USER;
      const pass = process.env.IPROYAL_PROXY_PASS;
      if (!user || !pass) return null;
      const sid = opts.sessionId ?? Math.random().toString(36).slice(2, 10);
      const ttl = opts.sessionTtlSec ?? 1800;     // 30 min sticky per RESEARCH.md
      const userPart = `${user}-country-${opts.geo}-session-${sid}-sessionttl-${ttl}`;
      return `http://${userPart}:${pass}@geo.iproyal.com:12321`;
    }
    ```

    ```typescript
    import { PlaywrightCrawler, type ProxyConfiguration, ProxyConfiguration as PCO } from 'crawlee';
    import { firefox } from 'playwright';
    import { eq, sql } from 'drizzle-orm';
    import type { Database } from '../../infra/db.js';
    import type { Queue } from '../../infra/queue.js';
    import { cars } from '../../db/schema/index.js';
    import { rehostImages } from './shared/images.js';
    import { detectBlock, BlockCounter, isPaused } from './shared/block-detection.js';
    import { parseListingPage, normalizeEncar } from './encar/normalize.js';
    import { buildProxyUrl } from '../../infra/proxy.js';

    export type EncarScrapeOpts = { db: Database; queue: Queue; mode?: 'dry-run' | 'live'; maxListings?: number };

    export async function runEncarScrape(opts: EncarScrapeOpts): Promise<{ status: 'success'|'partial'|'failed'|'blocked'; pages_fetched: number; cars_seen: number; cars_added: number; cars_updated: number; images_rehosted: number; block_signal?: string }> {
      if (await isPaused('encar', opts.db)) return { status: 'success', pages_fetched: 0, cars_seen: 0, cars_added: 0, cars_updated: 0, images_rehosted: 0 };

      const counter = new BlockCounter('encar', opts.db, opts.queue);
      const counters = { pages_fetched: 0, cars_seen: 0, cars_added: 0, cars_updated: 0, images_rehosted: 0 };
      const proxyUrl = opts.mode === 'dry-run' ? null : buildProxyUrl({ geo: 'kr', sessionTtlSec: 1800 });
      const proxyConfiguration = proxyUrl
        ? new PCO({ proxyUrls: [proxyUrl] })
        : undefined;

      const crawler = new PlaywrightCrawler({
        launchContext: { launcher: firefox, launchOptions: { headless: true } },
        proxyConfiguration,
        maxConcurrency: 1,
        maxRequestsPerCrawl: opts.maxListings ?? 150,
        navigationTimeoutSecs: 60,
        requestHandlerTimeoutSecs: 90,
        browserPoolOptions: {
          useFingerprints: true,
          maxOpenPagesPerBrowser: 1,
          retireBrowserAfterPageCount: 50,
        },
        async requestHandler({ page, request, log, enqueueLinks }) {
          await page.setExtraHTTPHeaders({ 'Accept-Language': 'ko-KR,ko;q=0.9' });
          await page.waitForLoadState('domcontentloaded');
          const html = await page.content();
          counters.pages_fetched++;
          const block = detectBlock({ html, status: 200 });
          const halted = (await counter.record(block)).halted;
          if (halted) throw new Error('BLOCKED:auto-halt');

          // Branch by URL shape: listing page vs detail page
          if (/dc_carsearchlist\.do/.test(request.url)) {
            const ids = parseListingPage(html);
            counters.cars_seen += ids.length;
            for (const { sourceId, sourceUrl } of ids) {
              await crawler.requestQueue?.addRequest({ url: sourceUrl, userData: { kind: 'detail', sourceId } });
            }
          } else if (request.userData?.kind === 'detail') {
            const sourceId = request.userData.sourceId as string;
            const normalized = await normalizeEncar(sourceId, request.url, html, opts.db);
            const imageKeys = await rehostImages(normalized.imageUrls, 'encar', sourceId);
            counters.images_rehosted += imageKeys.length;

            // UPSERT
            const result = await opts.db.insert(cars).values({
              source: 'encar', sourceId, sourceUrl: request.url,
              brand: normalized.brand, model: normalized.model, year: normalized.year,
              country: 'kr', body: normalized.body, drive: normalized.drive,
              fuel: normalized.fuel, transmission: normalized.transmission,
              mileageKm: normalized.mileageKm, priceLocal: normalized.priceLocal,
              priceLocalCcy: 'KRW', specSummary: normalized.specSummary,
              imageKey: imageKeys[0] ?? null,
              firstSeenAt: new Date(), lastSeenAt: new Date(),
              isActive: true, needsReview: true,
            }).onConflictDoUpdate({
              target: [cars.source, cars.sourceId],
              set: {
                brand: normalized.brand, model: normalized.model, year: normalized.year,
                priceLocal: normalized.priceLocal, mileageKm: normalized.mileageKm,
                imageKey: imageKeys[0] ?? null, lastSeenAt: sql`NOW()`,
                isActive: true, updatedAt: sql`NOW()`,
              },
            }).returning({ id: cars.id, isNew: sql<boolean>`xmax = 0` });

            if (result[0]?.isNew) counters.cars_added++; else counters.cars_updated++;
          }
        },
      });

      try {
        const seedUrl = 'https://www.encar.com/dc/dc_carsearchlist.do?carType=for';
        await crawler.run([seedUrl]);
        return { status: 'success', ...counters };
      } catch (e: any) {
        if (String(e?.message ?? '').startsWith('BLOCKED:')) {
          return { status: 'blocked', ...counters, block_signal: 'thin_or_captcha' };
        }
        return { status: 'failed', ...counters };
      } finally {
        await crawler.teardown();    // SCRAPE-08 isolation
      }
    }
    ```

    ```typescript
      // Split on / ; if EN half present, parse year + brand + model from it (more reliable).

      const modelWord = enWords[2] ?? '';

      const brand = aliasBrand?.cyrillicForm ?? aliasBrand?.latinForm ?? brandWord;
      // KRW price — Carapis returns major units already

      return {
        source: 'encar', sourceId: String(rec.id), sourceUrl: rec.url ?? `https://www.encar.com/dc/dc_carsearchview.do?carid=${rec.id}`,

        transmission: rec.specifications?.transmission ?? null,
    ```
