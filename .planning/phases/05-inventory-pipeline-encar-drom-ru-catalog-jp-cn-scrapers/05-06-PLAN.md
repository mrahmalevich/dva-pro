    /** Detail page parser → raw dict. */

      year: number; brand: string; model: string; grade: string;
      bodyEn: string | null; driveEn: string | null; fuelEn: string | null; transmissionEn: string | null;
      mileageRaw: string | null; priceJpy: string | null; imageUrls: string[];
    } { /* ... */ }

    export async function normalizeBeForward(
      sourceId: string,
      sourceUrl: string,
      detailHtml: string,
      db: Database
    ): Promise<NormalizedCar> {
      const raw = parseDetailPage(detailHtml);
      const aliasBrand = await lookupBrandAlias(db, { kind: 'brand', latinForm: raw.brand });

      const model = raw.model;

      return {
        source: 'beforward',
        sourceId,
        sourceUrl,
        brand, model,
        year: raw.year,
        country: 'jp',
        body: mapBody(raw.bodyEn ?? ''),
        drive: mapDrive(raw.driveEn ?? ''),
        fuel: mapFuel(raw.fuelEn ?? ''),
        transmission: mapTransmission(raw.transmissionEn ?? ''),
        mileageKm: parseMileage(raw.mileageRaw ?? ''),
        priceLocal: parsePriceJp(raw.priceJpy ?? '') ?? 0n,
        priceLocalCcy: 'JPY',
        specSummary: buildSpecSummary({ /* compute */ }),
        imageUrls: raw.imageUrls,
        needsReview: true,
      };
    }

    ```typescript
    import { CheerioCrawler } from 'crawlee';
    import { sql } from 'drizzle-orm';
    import type { Database } from '../../infra/db.js';
    import type { Queue } from '../../infra/queue.js';
    import { cars } from '../../db/schema/index.js';
    import { rehostImages } from './shared/images.js';
    import { detectBlock, BlockCounter, isPaused } from './shared/block-detection.js';
    import { parseListingPage, normalizeBeForward } from './beforward/normalize.js';

    export type BeForwardScrapeOpts = { db: Database; queue: Queue; mode?: 'dry-run' | 'live'; maxListings?: number };

    export async function runBeForwardScrape(opts: BeForwardScrapeOpts) {
      if (await isPaused('beforward', opts.db)) return { status: 'success' as const, pages_fetched: 0, cars_seen: 0, cars_added: 0, cars_updated: 0, images_rehosted: 0 };

      const counter = new BlockCounter('beforward', opts.db, opts.queue);
      const counters = { pages_fetched: 0, cars_seen: 0, cars_added: 0, cars_updated: 0, images_rehosted: 0 };

      const crawler = new CheerioCrawler({
        // NO proxyConfiguration — D-04 RU DC direct egress
        maxConcurrency: 1,
        maxRequestsPerCrawl: opts.maxListings ?? 200,
        navigationTimeoutSecs: 30,
        async requestHandler({ $, request, body }) {
          const html = String(body);
          counters.pages_fetched++;
          const block = detectBlock({ html, status: 200 });
          if ((await counter.record(block)).halted) throw new Error('BLOCKED:auto-halt');

          if (request.userData?.kind === 'detail') {
            const sourceId = request.userData.sourceId as string;
            const normalized = await normalizeBeForward(sourceId, request.url, html, opts.db);
            const imageKeys = await rehostImages(normalized.imageUrls, 'beforward', sourceId);
            counters.images_rehosted += imageKeys.length;

            const result = await opts.db.insert(cars).values({
              source: 'beforward', sourceId, sourceUrl: request.url,
              brand: normalized.brand, model: normalized.model, year: normalized.year,
              country: 'jp', body: normalized.body, drive: normalized.drive,
              fuel: normalized.fuel, transmission: normalized.transmission,
              mileageKm: normalized.mileageKm, priceLocal: normalized.priceLocal,
              priceLocalCcy: 'JPY', specSummary: normalized.specSummary,
              imageKey: imageKeys[0] ?? null,
              firstSeenAt: new Date(), lastSeenAt: new Date(),
              isActive: true, needsReview: true,
            }).onConflictDoUpdate({
              target: [cars.source, cars.sourceId],
              set: {
                brand: normalized.brand, model: normalized.model,
                priceLocal: normalized.priceLocal, mileageKm: normalized.mileageKm,
                imageKey: imageKeys[0] ?? null, lastSeenAt: sql`NOW()`,
                isActive: true, updatedAt: sql`NOW()`,
              },
            }).returning({ id: cars.id, isNew: sql<boolean>`xmax = 0` });

            if (result[0]?.isNew) counters.cars_added++; else counters.cars_updated++;
          } else {
            // Listing page: parse refs and enqueue detail URLs
            const ids = parseListingPage(html);
            counters.cars_seen += ids.length;
            for (const { sourceId, sourceUrl } of ids) {
              await crawler.requestQueue?.addRequest({ url: sourceUrl, userData: { kind: 'detail', sourceId } });
            }
          }
        },
      });

      try {
        const seedUrl = 'https://www.beforward.jp/stocklist/?currency=JPY';
        await crawler.run([seedUrl]);
        return { status: 'success' as const, ...counters };
      } catch (e: any) {
        if (String(e?.message ?? '').startsWith('BLOCKED:')) return { status: 'blocked' as const, ...counters, block_signal: 'thin_or_captcha' };
        return { status: 'failed' as const, ...counters };
      } finally {
        await crawler.teardown();
      }
    }
    ```
