    1. **`extractCellValue(td)` — value-cell normaliser:**
       ```ts
       /**
        * Normalise a value <td> from a comp data row.
        * Order of checks matters: SVG → boolean; «опция» span → string; numeric+unit → number; else trimmed text.
        */
       function extractCellValue(td: cheerio.Cheerio<cheerio.Element>): string | boolean | number | null {
         // 1. yes/no SVG (drom uses <use xlink:href="#yes"|"#no">)
         const useEl = td.find('use[xlink\\:href="#yes"], use[xlink\\:href="#no"]').first();
         if (useEl.length > 0) {
           const href = useEl.attr('xlink:href') ?? '';
           if (href === '#yes') return true;
           if (href === '#no') return false;
         }
         // 2. «опция» badge — drom wraps it in nested spans; look for any descendant whose trimmed text === 'опция'
         const optionSpan = td.find('span').filter((_, el) => td.find(el).text().trim() === 'опция').first();
         if (optionSpan.length > 0) return 'опция';
         // 3. text — trim whitespace and strip nested element noise
         //    Take the first text node OR the full text.trim() if the cell has nested HTML.
         const raw = td.text().replace(/\\s+/g, ' ').trim();
         if (!raw) return null;
         // 4. numeric? Match leading number with optional decimal (comma or dot).
         //    Reject if the WHOLE string is non-numeric followed by units — e.g. "плавающие диски" stays string.
         //    Pattern: optional minus, digits, optional [.,]digits, then optional whitespace + unit text.
         const numMatch = raw.match(/^(-?\\d+(?:[.,]\\d+)?)(\\s|$)/);
         if (numMatch) {
           const n = Number(numMatch[1].replace(',', '.'));
           if (Number.isFinite(n)) return n;
         }
         return raw;
       }
       ```

    2. **`walkSections($)` — generic walker:**
       ```ts
       interface WalkedRow {
         section: string;
         subsection: string | null;
         label: string;
         normalisedValue: string | boolean | number | null;
       }

       function walkSections($: cheerio.CheerioAPI): WalkedRow[] {
         const rows: WalkedRow[] = [];
         let currentSection: string | null = null;
         let currentSubsection: string | null = null;

         // Walk every <tr> within the comp specs table.
         // The relevant table is the one with class `bm-catCompTableWrap` (a wrapping div) > table > tbody > tr.
         // Use a broad selector then filter so we don't depend on the wrapper class staying stable.
         $('table.b-table tr').each((_, tr) => {
           const $tr = $(tr);
           const cls = $tr.attr('class') ?? '';
           const isSection = /\\bb-text_size_xxl\\b/.test(cls);
           const isSubsection = /\\bb-text_size_xl\\b/.test(cls) && !isSection;
           const isDataRow = /\\bb-table__row_cols_2\\b/.test(cls);

           if (isSection) {
             currentSection = $tr.find('td').first().text().replace(/\\s+/g, ' ').trim();
             currentSubsection = null;
             return;
           }
           if (isSubsection) {
             currentSubsection = $tr.find('td').first().text().replace(/\\s+/g, ' ').trim();
             return;
           }
           if (isDataRow && currentSection) {
             const tds = $tr.find('> td');
             if (tds.length < 2) return;
             const label = $(tds[0]).text().replace(/\\s+/g, ' ').trim();
             if (!label) return;
             const value = extractCellValue($(tds[1]));
             rows.push({
               section: currentSection,
               subsection: currentSubsection,
               label,
               normalisedValue: value,
             });
           }
         });
         return rows;
       }
       ```

      matcher: RegExp;                                              // tested against label.toLowerCase()
      path: readonly [keyof Complectation, string];                 // ['drivetrain', 'turbo'] etc

    const asBool = (v: unknown) => (typeof v === 'boolean' ? v : null);
    const asNum = (v: unknown) => (typeof v === 'number' ? v : null);

    Then add a helper:

    ```ts
    /**
     * Apply the walked rows to (a) populate typed slots in `slots` and (b) push remaining rows into `features`.
     * Mutates both `slots` and `features` in place. Used by parseComplectationPage.
     */
    function applyWalkedRows(
      walked: ReadonlyArray<WalkedRow>,
      slots: { drivetrain: Complectation['drivetrain']; dimensions: Complectation['dimensions']; chassis: Complectation['chassis'] },
      features: Complectation['features'],
    ): void {
      for (const row of walked) {
        const labelLower = row.label.toLowerCase();
        // Try each mapping; first match wins.
        let matched = false;
        for (const m of TYPED_SLOT_MAPPINGS) {
          if (m.matcher.test(labelLower)) {
            const coerced = m.coerce(row.normalisedValue);
            if (coerced !== null) {
              const [group, leaf] = m.path;
              // Type-safe assignment via record cast — slots groups are typed objects.
              (slots[group] as Record<string, unknown>)[leaf] = coerced;
            }
            matched = true;
            break;
          }
        }
        if (matched) continue;
        // Fallback: push to features bag if value is a usable type.
        if (isFeatureValue(row.normalisedValue)) {
          features.push({
            section: row.section,
            subsection: row.subsection,
            label: row.label,
            value: row.normalisedValue,
          });
        }
        // null normalisedValue → drop silently (empty cell)
      }
    }
    ```
