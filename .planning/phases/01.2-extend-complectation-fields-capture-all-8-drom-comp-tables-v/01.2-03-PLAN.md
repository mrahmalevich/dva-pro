    ```javascript
    // ---- Phase 01.2: render 8 native drom sections from features[] ----
    //
    // features is an array of { section, subsection, label, value } entries.
    // We group preserving first-encountered order (drom's source order).
    // Booleans render as ✓/× glyphs; strings/numbers render verbatim through esc().
    function renderFeaturesSections(features) {
      if (!Array.isArray(features) || features.length === 0) return '';

      // Group: section → Map<subsection|'__null__', Array<feature>>, preserving order
      const sections = new Map();
      for (const f of features) {
        if (!f || typeof f.section !== 'string') continue;
        if (!sections.has(f.section)) sections.set(f.section, new Map());
        const subKey = f.subsection === null || f.subsection === undefined ? '__null__' : f.subsection;
        const subs = sections.get(f.section);
        if (!subs.has(subKey)) subs.set(subKey, []);
        subs.get(subKey).push(f);
      }

      let html = '';
      for (const [sectionName, subsections] of sections) {
        html += '<div class="modal-section"><h3>' + esc(sectionName) + '</h3>';
        for (const [subKey, entries] of subsections) {
          // Subsection sub-heading (skip the synthetic "__null__" key — render a flat list)
          if (subKey !== '__null__') {
            html += '<h4 style="color: var(--muted); font-size: 12px; margin: 12px 0 6px; text-transform: none; letter-spacing: 0;">' + esc(subKey) + '</h4>';
          }
          html += '<dl class="kv">';
          for (const entry of entries) {
            const label = typeof entry.label === 'string' ? entry.label : '';
            if (!label) continue;
            let valueHtml;
            if (entry.value === true) {
              valueHtml = '<span style="color: var(--ok);">✓</span>';
            } else if (entry.value === false) {
              valueHtml = '<span style="color: var(--err);">×</span>';
            } else if (typeof entry.value === 'number') {
              valueHtml = esc(String(entry.value));
            } else if (typeof entry.value === 'string') {
              valueHtml = esc(entry.value);
            } else {
              continue; // unknown type — skip
            }
            html += '<dt>' + esc(label) + '</dt><dd>' + valueHtml + '</dd>';
          }
          html += '</dl>';
        }
        html += '</div>';
      }
      return html;
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| features[].label / features[].value (drom) → embedded HTML | Untrusted strings flow into the operator-facing index.html via JSON.stringify into a `<script type="application/json">` tag, then into `innerHTML` via the modal renderer |
| operator browser (file://) → DOM | Rendered HTML opens via `file://` in the operator's browser — no CORS, but XSS in features could execute arbitrary JS in the same origin as the local file |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01.2-08 | Tampering / XSS | features[].label or features[].value containing HTML/script payload | mitigate | All string values flow through the existing `esc()` helper in the embedded script (escapes &<>"'). Booleans never reach esc(); they are emitted as fixed `<span>...</span>` literals. Numbers go through `esc(String(n))` defensively. |
| T-01.2-09 | Tampering | features[].section or features[].subsection containing HTML | mitigate | Section + subsection names are emitted only via `esc()`. Confirmed by Task 2 test 6 (XSS payload renders escaped). |
| T-01.2-10 | Information Disclosure | Operator viewer leaks per-comp data | accept | Viewer is local-only (file://); no upload path; data is public drom catalog content. |
</threat_model>
